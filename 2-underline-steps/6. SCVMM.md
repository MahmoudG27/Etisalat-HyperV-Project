# Create svc-vmm user This is the account that the VMM service uses to operate in the background. like "ServiceAccount" `www-data` for nginx not Root user.
```powershell
Import-Module ActiveDirectory

# User details
$Username = "svc-vmm"
$DomainDNS = "etisalat.local"
$DomainDN = (Get-ADDomain -Identity $DomainDNS).DistinguishedName
$Password = Read-Host "Enter password for svc-vmm" -AsSecureString

# Create the service account
New-ADUser -Name $Username -SamAccountName $Username -UserPrincipalName "$Username@$DomainDNS" -Description "Service Account for Virtual Machine Manager (VMM) Administration" -Path "CN=Users,$DomainDN" -AccountPassword $Password -Enabled $true -PasswordNeverExpires $true -ChangePasswordAtLogon $false

# Add the account to Domain Admins
Add-ADGroupMember -Identity "Domain Admins" -Members $Username

# Verify the account
Get-ADUser $Username -Properties Description, PasswordNeverExpires, Enabled | Select-Object Name, SamAccountName, Description, PasswordNeverExpires, Enabled

# Verify group membership
Get-ADPrincipalGroupMembership $Username | Select-Object Name
```

## Install SCVMM:

# Create VM for VMM's SQL.
```powershell
New-VM -Name "SQL01" -Generation 2 -MemoryStartupBytes 16GB -NewVHDPath "D:\VMs\SQL01\SQL01.vhdx" -NewVHDSizeBytes 200GB -SwitchName "Management-vSwitch"
Set-VMProcessor SQL01 -Count 4
Set-VMMemory -VMName "SQL01" -DynamicMemoryEnabled $false -StartupBytes 16GB
Add-VMDvdDrive -VMName "SQL01" -Path "C:\ISO\WindowsServer2025.iso"

$DVD = Get-VMDvdDrive -VMName "SQL01"
Set-VMFirmware -VMName "SQL01" -FirstBootDevice $DVD

Start-VM SQL01
```
